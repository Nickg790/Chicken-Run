/* ==========================================================================
   Chicken Run â€” client
   Canvas renderer + deterministic tile engine + hop physics + touch controls
   + cashout scoring + Socket.io match sync.

   Determinism contract with the server:
     The server hands both clients the same `seed` and a `startAt` timestamp on
     the shared (clock-synced) timeline. Lanes are generated by a seeded PRNG in
     strict row order, and every moving obstacle's position is a pure function
     of match time t. Both players race a byte-identical map, while only a few
     numbers per player cross the wire at 20 Hz.

   Scoring contract:
     Everything you earn is PENDING and at risk. CASHOUT banks it and ends your
     run. DEATH forfeits all of it â€” nothing is ever auto-saved on death.
     Players never collide with each other; the rival is a ghost, not a body.
   ========================================================================== */

   */

  var SOCKET_URL = (function () {
    var h = window.location.hostname;
    if (h === 'localhost' || h === '127.0.0.1' || h === '') return 'http://localhost:3000';
    if (/\.onrender\.com$/.test(h)) return window.location.origin;   // served by the API itself
    return RENDER_URL;
  })();

  function clamp(v, lo, hi) { return v < lo ? lo : (v > hi ? hi : v); }
  function lerp(a, b, t) { return a + (b - a) * t; }
  function mod(a, n) { return ((a % n) + n) % n; }
  function easeOutCubic(t) { return 1 - Math.pow(1 - t, 3); }

  // mulberry32 â€” tiny, fast, and identical across engines.
  function mulberry32(a) {
    return function () {
      a |= 0; a = (a + 0x6D2B79F5) | 0;
      var t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  function hashString(s) {
    var h = 2166136261 >>> 0;
    for (var i = 0; i < s.length; i++) {
      h ^= s.charCodeAt(i);
      h = Math.imul(h, 16777619);
    }
    return h >>> 0;
  }

  function rr(ctx, x, y, w, h, r) {
    r = Math.min(r, Math.abs(w) / 2, Math.abs(h) / 2);
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.lineTo(x + w - r, y);
    ctx.quadraticCurveTo(x + w, y, x + w, y + r);
    ctx.lineTo(x + w, y + h - r);
    ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    ctx.lineTo(x + r, y + h);
    ctx.quadraticCurveTo(x, y + h, x, y + h - r);
    ctx.lineTo(x, y + r);
    ctx.quadraticCurveTo(x, y, x + r, y);
    ctx.closePath();
  }

  function ellipse(ctx, cx, cy, rx, ry) {
    ctx.beginPath();
    ctx.ellipse(cx, cy, Math.max(0.5, rx), Math.max(0.5, ry), 0, 0, Math.PI * 2);
  }

  /* ====================================================================== */
  /* Lifetime head-to-head record (localStorage)                             */
  /* ====================================================================== */

  var H2H = {
    read: function (k, dflt) {
      try { var v = localStorage.getItem(k); return v === null ? dflt : v; }
      catch (e) { return dflt; }
    },
    write: function (k, v) {
      try { localStorage.setItem(k, String(v)); } catch (e) {}
    },

    myName: function () { return this.read('cr_name', '') || 'YOU'; },
    setMyName: function (n) { this.write('cr_name', n); },

    lastRival: function () { return this.read('cr_last_rival', ''); },
    setLastRival: function (n) { if (n) this.write('cr_last_rival', n); },

    allTimeBest: function () { return parseInt(this.read('cr_best', '0'), 10) || 0; },
    noteBest: function (score) {
      var b = this.allTimeBest();
      if (score > b) { this.write('cr_best', score); return score; }
      return b;
    },

    key: function (rival) {
      return 'cr_h2h_' + String(rival || 'RIVAL').toUpperCase().replace(/[^A-Z0-9]/g, '_');
    },

    get: function (rival) {
      var raw = this.read(this.key(rival), null);
      var rec = { w: 0, l: 0, d: 0, myBest: 0, rivalBest: 0 };
      if (raw) {
        try {
          var p = JSON.parse(raw);
          rec.w = p.w | 0; rec.l = p.l | 0; rec.d = p.d | 0;
          rec.myBest = p.myBest | 0; rec.rivalBest = p.rivalBest | 0;
        } catch (e) {}
      }
      return rec;
    },

    save: function (rival, rec) {
      try { this.write(this.key(rival), JSON.stringify(rec)); } catch (e) {}
    },

    /** Fold one finished match into the lifetime record for this pairing. */
    record: function (rival, result, myBanked, rivalBanked) {
      var rec = this.get(rival);
      if (result === 'win') rec.w++;
      else if (result === 'loss') rec.l++;
      else rec.d++;
      if (myBanked > rec.myBest) rec.myBest = myBanked;
      if (rivalBanked > rec.rivalBest) rec.rivalBest = rivalBanked;
      this.save(rival, rec);
      this.setLastRival(rival);
      this.noteBest(myBanked);
      return rec;
    }
  };

  /* ====================================================================== */
  /* Sound â€” tiny WebAudio blips, no assets                                  */
  /* ====================================================================== */

  var SFX = {
    ctx: null,
    on: true,

    init: function () {
      this.on = H2H.read('cr_sound', '1') !== '0';
    },
    setOn: function (v) { this.on = v; H2H.write('cr_sound', v ? '1' : '0'); },

    ensure: function () {
      if (!this.on) return null;
      try {
        if (!this.ctx) {
          var AC = window.AudioContext || window.webkitAudioContext;
          if (!AC) return null;
          this.ctx = new AC();
        }
        if (this.ctx.state === 'suspended') this.ctx.resume();
        return this.ctx;
      } catch (e) { return null; }
    },

    blip: function (freq, dur, type, gain, slideTo) {
      var c = this.ensure();
      if (!c) return;
      try {
        var o = c.createOscillator();
        var g = c.createGain();
        o.type = type || 'square';
        o.frequency.setValueAtTime(freq, c.currentTime);
        if (slideTo) o.frequency.exponentialRampToValueAtTime(slideTo, c.currentTime + dur);
        g.gain.setValueAtTime(gain == null ? 0.05 : gain, c.currentTime);
        g.gain.exponentialRampToValueAtTime(0.0001, c.currentTime + dur);
        o.connect(g); g.connect(c.destination);
        o.start(); o.stop(c.currentTime + dur + 0.02);
      } catch (e) {}
    },

    hop:   function () { this.blip(520, 0.07, 'square', 0.030, 700); },
    ping:  function () { this.blip(880, 0.10, 'triangle', 0.055, 1320); },
    near:  function () { this.blip(300, 0.09, 'sawtooth', 0.030, 190); },
    death: function () { this.blip(220, 0.34, 'sawtooth', 0.070, 60); },
    cash:  function () {
      var self = this;
      [660, 880, 1180].forEach(function (f, i) {
        setTimeout(function () { self.blip(f, 0.13, 'triangle', 0.060); }, i * 85);
      });
    }
  };

  /* ====================================================================== */
  /* World â€” deterministic lane generation                                   */
  /* ====================================================================== */

  var PAD = 8; // rows below 0 we still generate (start runway)

  var World = {
    seed: 0,
    rng: null,
    lanes: [],
    runType: null,
    runLeft: 0,
    lastHazard: false,
    lastRailRow: -99,
    hazardStreak: 0,
    padCorridor: null,

    reset: function (seedStr) {
      this.seed = hashString(String(seedStr));
      this.rng = mulberry32(this.seed);
      this.lanes = [];
      this.runType = null;
      this.runLeft = 0;
      this.lastHazard = false;
      this.lastRailRow = -99;
      this.hazardStreak = 0;
      this.padCorridor = null;
      this.ensure(CFG.SAFE_ROWS + 20);
    },

    ensure: function (upToRow) {
      while (this.lanes.length <= upToRow + PAD) {
        var row = this.lanes.length - PAD;
        this.lanes.push(this.generate(row));
      }
    },

    get: function (row) {
      this.ensure(row + 12);
      var idx = row + PAD;
      if (idx < 0) idx = 0;
      return this.lanes[idx];
    },

    pickType: function (row) {
      var r = this.rng();
      // A hazard block is usually followed by a breather of grass.
      if (this.lastHazard && r < 0.40) return 'grass';
      if (r < 0.40) return 'road';
      if (r < 0.68) return 'grass';
      if (r < 0.88) return 'river';
      // Railways need elbow room. Left unchecked the run-length picker can
      // stack seven tracks into one screen, which is both a wall of crossing
      // signals and a stretch you simply cannot cross.
      if (row - this.lastRailRow < 6) return this.rng() < 0.5 ? 'grass' : 'road';
      return 'rail';
    },

    generate: function (row) {
      var rng = this.rng;
      var type;

      // The map is endless â€” a run ends by CASHOUT or DEATH, never by a line.
      if (row < CFG.SAFE_ROWS) {
        type = 'grass';
      } else if (this.hazardStreak >= CFG.MAX_HAZARD_RUN) {
        // Hard ceiling on unbroken danger. Run-lengths alone can chain river
        // into river into river until thirteen of the fourteen rows on screen
        // are water â€” technically survivable, actually miserable.
        this.runType = 'grass';
        this.runLeft = 1 + ((rng() * 2) | 0);
        type = 'grass';
        this.runLeft--;
        this.lastHazard = false;
      } else {
        if (this.runLeft <= 0) {
          this.runType = this.pickType(row);
          if (this.runType === 'grass') this.runLeft = 1 + ((rng() * 2) | 0);
          else if (this.runType === 'road') this.runLeft = 1 + ((rng() * 3) | 0);
          else if (this.runType === 'river') this.runLeft = 1 + ((rng() * 3) | 0);
          else this.runLeft = 1 + ((rng() * 2) | 0);
          this.lastHazard = this.runType !== 'grass';
        }
        type = this.runType;
        this.runLeft--;
      }

      this.hazardStreak = (type === 'grass') ? 0 : this.hazardStreak + 1;

      if (type === 'grass') return this.makeGrass(row);
      if (type === 'road') return this.makeRoad(row);
      if (type === 'river') return this.makeRiver(row);
      this.lastRailRow = row;
      return this.makeRail(row);
    },

    makeGrass: function (row) {
      var rng = this.rng;
      var lane = { row: row, type: 'grass', shade: row & 1, trees: {} };
      var startZone = row < CFG.SAFE_ROWS;
      var treeChance = startZone ? 0.04 : 0.15;
      var open = 0;

      for (var x = CFG.MIN_X; x <= CFG.MAX_X; x++) {
        var protectedTile = startZone && Math.abs(x) <= 1;
        if (!protectedTile && rng() < treeChance) {
          // Rocks and trees are the same thing to the game â€” a blocked tile.
          // Splitting them only varies the sprite, so density is unchanged.
          lane.trees[x] = { s: 0.95 + rng() * 0.35, rock: rng() < 0.26 };
        } else {
          open++;
        }
      }
      // Never let a row wall itself off.
      if (open < 4) lane.trees = {};
      return lane;
    },

    makeRoad: function (row) {
      var rng = this.rng;
      var kind = rng();
      var vType, len, speed, colors;
      // Traffic is deliberately unhurried â€” the tension is supposed to come
      // from deciding when to cash out, not from twitch reflexes.
      if (kind < 0.58) {
        vType = 'car';   len = 1.55; speed = 1.65 + rng() * 1.05; colors = CAR_COLORS;
      } else if (kind < 0.84) {
        vType = 'truck'; len = 2.55; speed = 1.30 + rng() * 0.70; colors = TRUCK_COLORS;
      } else {
        vType = 'bus';   len = 3.15; speed = 1.05 + rng() * 0.55; colors = BUS_COLORS;
      }

      var dir = rng() < 0.5 ? 1 : -1;
      var gap = len + 3.4 + rng() * 4.0;
      var span = Math.max(CFG.SPAWN_R - CFG.SPAWN_L, gap * 2);
      var count = Math.max(2, Math.ceil(span / gap));
      span = count * gap;

      var items = [];
      for (var i = 0; i < count; i++) items.push({ color: colors[(rng() * colors.length) | 0] });

      return {
        row: row, type: 'road', shade: row & 1, vType: vType,
        dir: dir, speed: speed, len: len, gap: gap, span: span,
        offset: rng() * span, items: items
      };
    },

    makeRiver: function (row) {
      var rng = this.rng;

      // A river lane is EITHER drifting logs or static lily pads â€” never both.
      // Pads can't fill the gaps of a log lane because those gaps move with the
      // logs, so a fixed pad would spend half its life underneath one.
      if (rng() < 0.42) return this.makePadRiver(row);

      var dir = rng() < 0.5 ? 1 : -1;
      var speed = 0.9 + rng() * 1.35;
      var gap = 4.0 + rng() * 4.2;
      var span = Math.max(CFG.SPAWN_R - CFG.SPAWN_L, gap * 2);
      var count = Math.max(2, Math.ceil(span / gap));
      span = count * gap;

      var items = [];
      for (var i = 0; i < count; i++) {
        items.push({ len: 2 + Math.floor(rng() * 3) + (rng() < 0.3 ? 0.5 : 0) });
      }
      return {
        row: row, type: 'river', mode: 'logs', dir: dir, speed: speed,
        gap: gap, span: span, offset: rng() * span, items: items, pads: null
      };
    },

    /**
     * Stepping-stone river: lily pads sit still and hold your weight.
     * A 3-wide corridor that drifts by at most one column per lane guarantees
     * consecutive pad lanes always overlap, so a crossing always exists.
     */
    makePadRiver: function (row) {
      var rng = this.rng;
      if (this.padCorridor === null) {
        this.padCorridor = ((rng() * 5) | 0) - 2;
      } else {
        this.padCorridor = clamp(this.padCorridor + (((rng() * 3) | 0) - 1),
                                 CFG.MIN_X + 1, CFG.MAX_X - 1);
      }

      var pads = {};
      var mk = function () {
        return { r: 0.92 + rng() * 0.38, a: rng() * Math.PI * 2, flower: rng() < 0.18 };
      };
      for (var d = -1; d <= 1; d++) pads[this.padCorridor + d] = mk();
      for (var x = CFG.MIN_X; x <= CFG.MAX_X; x++) {
        if (!pads[x] && rng() < 0.30) pads[x] = mk();
      }

      return {
        row: row, type: 'river', mode: 'pads',
        dir: rng() < 0.5 ? 1 : -1,     // visual ripple direction only
        speed: 0.30,                   // visual ripple speed only
        gap: 0, span: 1, offset: 0,
        items: [], pads: pads
      };
    },

    makeRail: function (row) {
      var rng = this.rng;
      return {
        row: row, type: 'rail',
        dir: rng() < 0.5 ? 1 : -1,
        period: 4.6 + rng() * 3.2,
        phase: rng() * 8
      };
    }
  };

  /* ---------------------------------------------------------------------- */
  /* Obstacle position math â€” pure functions of match time                   */
  /* ---------------------------------------------------------------------- */

  function wrapX(v, span) { return mod(v - CFG.SPAWN_L, span) + CFG.SPAWN_L; }

  /** Centre x of moving item i on a road/river lane at match time t. */
  function itemX(lane, i, t) {
    return wrapX(lane.offset + i * lane.gap + lane.dir * lane.speed * t, lane.span);
  }

  /** Train state for a rail lane: {warn, active, head, tail}. */
  function trainState(lane, t) {
    var tc = mod(t + lane.phase, lane.period);
    var warn = tc < CFG.TRAIN_WARN;
    var out = { warn: warn, active: false, head: 0, tail: 0, blink: (tc * 6.5) % 1 < 0.5 };
    if (tc >= CFG.TRAIN_WARN && tc < CFG.TRAIN_WARN + CFG.TRAIN_SWEEP) {
      var p = (tc - CFG.TRAIN_WARN) / CFG.TRAIN_SWEEP;
      var travel = (CFG.SPAWN_R - CFG.SPAWN_L) + CFG.TRAIN_LEN * 2;
      out.active = true;
      if (lane.dir > 0) {
        out.head = CFG.SPAWN_L - CFG.TRAIN_LEN + p * travel;
        out.tail = out.head - CFG.TRAIN_LEN;
      } else {
        out.head = CFG.SPAWN_R + CFG.TRAIN_LEN - p * travel;
        out.tail = out.head + CFG.TRAIN_LEN;
      }
    }
    return out;
  }

  /**
   * Whatever is holding the player up on a river lane, or null if that's open
   * water. Logs drift and carry you; lily pads are fixed and don't.
   */
  function platformUnder(lane, x, t) {
    if (lane.mode === 'pads') {
      var xi = Math.round(x);
      return lane.pads[xi] ? { pad: true, cx: xi } : null;
    }
    for (var i = 0; i < lane.items.length; i++) {
      var cx = itemX(lane, i, t);
      var half = lane.items[i].len / 2;
      if (x >= cx - half - 0.28 && x <= cx + half + 0.28) {
        return { pad: false, i: i, cx: cx, len: lane.items[i].len };
      }
    }
    return null;
  }

  var Popups = {
    list: [],

    add: function (text, color, opts) {
      opts = opts || {};
      this.list.push({
        text: text,
        color: color || '#fff',
        t: 0,
        life: opts.life || 0.95,
        size: opts.size || 0.36,       // multiple of tile size
        vy: opts.vy || 1.5,            // tiles per second upward
        drift: (Math.random() - 0.5) * 0.5,
        worldX: opts.worldX == null ? Player.fx : opts.worldX,
        worldRow: opts.worldRow == null ? Player.frow : opts.worldRow,
        centre: !!opts.centre
      });
      if (this.list.length > 24) this.list.shift();
    },

    update: function (dt) {
      for (var i = this.list.length - 1; i >= 0; i--) {
        var p = this.list[i];
        p.t += dt;
        if (p.t >= p.life) this.list.splice(i, 1);
      }
    },

    clear: function () { this.list.length = 0; },

    draw: function (ctx, View) {
      var T = View.tile;
      for (var i = 0; i < this.list.length; i++) {
        var p = this.list[i];
        var k = p.t / p.life;
        var rise = easeOutCubic(k) * p.vy * T;
        var alpha = k < 0.12 ? (k / 0.12) : (1 - Math.pow((k - 0.12) / 0.88, 2));
        var scale = k < 0.16 ? lerp(0.6, 1.12, k / 0.16) : lerp(1.12, 1, Math.min(1, (k - 0.16) / 0.3));

        var x, y;
        if (p.centre) {
          x = View.w / 2;
          y = View.h * 0.34 - rise;
        } else {
          // Popups live in screen space but track a board position, so the
          // anchor has to go through the same tilt the board does.
          var a = toScreen(sx(p.worldX + p.drift * k), sy(p.worldRow));
          x = a.x;
          y = a.y - T * 0.75 - rise;
        }

        ctx.save();
        ctx.globalAlpha = clamp(alpha, 0, 1);
        ctx.translate(x, y);
        ctx.scale(scale, scale);
        ctx.font = '900 ' + Math.round(T * p.size) + 'px ' + FONT;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.lineWidth = Math.max(2, T * 0.05);
        ctx.strokeStyle = 'rgba(6,10,16,0.8)';
        ctx.strokeText(p.text, 0, 0);
        ctx.fillStyle = p.color;
        ctx.shadowColor = p.color;
        ctx.shadowBlur = T * 0.16;
        ctx.fillText(p.text, 0, 0);
        ctx.restore();
      }
    }
  };

  var canvas = document.getElementById('game');
  var ctx = canvas.getContext('2d', { alpha: false });
  var View = { w: 0, h: 0, dpr: 1, tile: 40, padX: 0, padY: 0, cullL: 0, cullR: 0 };

  function resize() {
    var vv = window.visualViewport;
    var w = Math.round((vv && vv.width) || window.innerWidth);
    var h = Math.round((vv && vv.height) || window.innerHeight);
    // iPhone 16 / 16 Pro are 3x displays. Capping below 3 makes the canvas
    // resample and softens every edge on the exact devices this ships to, and
    // the board is flat fills the A18 eats for breakfast â€” so render native.
    var dpr = Math.min(window.devicePixelRatio || 1, 3);

    View.w = w; View.h = h; View.dpr = dpr;
    canvas.width = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    canvas.style.width = w + 'px';
    canvas.style.height = h + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.imageSmoothingEnabled = true;

    // Fit the requested column count, but keep tiles sane on tall/wide screens.
    View.tile = clamp(w / CFG.VIEW_COLS, 24, Math.max(32, h / 10));

    // Rotating the board sweeps its corners outside the viewport. The pivot is
    // the player's screen position, NOT the centre, so the top corners swing
    // further than a centred bounding box would suggest â€” pad from the pivot.
    var a = Math.abs(CFG.WORLD_ANGLE);
    var ca = Math.cos(a), sa = Math.sin(a);
    var py = h * CFG.PLAYER_SCREEN_Y;
    var reachX = 0, reachY = 0;
    [[-w / 2, -py], [w / 2, -py], [-w / 2, h - py], [w / 2, h - py]].forEach(function (c) {
      var dx = Math.abs(c[0]), dy = Math.abs(c[1]) / CFG.WORLD_SQUASH;
      reachX = Math.max(reachX, dx * ca + dy * sa);
      reachY = Math.max(reachY, dx * sa + dy * ca);
    });
    View.padX = Math.ceil(reachX - w / 2) + 12;
    View.padY = Math.ceil(reachY - Math.max(py, h - py)) + 12;
    View.cullL = -View.padX - View.tile * 3;
    View.cullR = w + View.padX + View.tile * 3;
  }

  window.addEventListener('resize', resize);
  window.addEventListener('orientationchange', function () { setTimeout(resize, 120); });
  if (window.visualViewport) window.visualViewport.addEventListener('resize', resize);
  resize();

  function sx(worldX) { return View.w / 2 + (worldX - Game.cam.x) * View.tile; }
  function sy(row) { return View.h * CFG.PLAYER_SCREEN_Y - (row - Game.cam.row) * View.tile; }

  /* Drawing helpers                                                         */
  /* ====================================================================== */

  /**
   * The whole playfield is drawn axis-aligned and then tilted as one unit, so
   * every gameplay calculation stays in clean tile space while the picture
   * reads as an angled isometric board. sx()/sy() give "board space"; toScreen()
   * maps that to final pixels for anything that must NOT tilt (text, signals).
   */
  function applyWorldTilt() {
    var cx = View.w / 2, cy = View.h * CFG.PLAYER_SCREEN_Y;
    ctx.translate(cx, cy);
    ctx.rotate(CFG.WORLD_ANGLE);
    ctx.scale(1, CFG.WORLD_SQUASH);
    ctx.translate(-cx, -cy);
  }

  function toScreen(px, py) {
    var cx = View.w / 2, cy = View.h * CFG.PLAYER_SCREEN_Y;
    var dx = px - cx, dy = (py - cy) * CFG.WORLD_SQUASH;
    var c = Math.cos(CFG.WORLD_ANGLE), s = Math.sin(CFG.WORLD_ANGLE);
    return { x: cx + dx * c - dy * s, y: cy + dx * s + dy * c };
  }

  /** Where a lane's centreline crosses a given final-screen x. */
  function laneEdgePoint(row, screenX) {
    var cx = View.w / 2, cy = View.h * CFG.PLAYER_SCREEN_Y;
    var c = Math.cos(CFG.WORLD_ANGLE), s = Math.sin(CFG.WORLD_ANGLE);
    var dy = (sy(row) - cy) * CFG.WORLD_SQUASH;
    var dx = (screenX - cx + dy * s) / c;
    return { x: screenX, y: cy + dx * s + dy * c };
  }

  function shade(hex, f) {
    var n = parseInt(hex.slice(1), 16);
    var r = Math.round(clamp(((n >> 16) & 255) * f, 0, 255));
    var g = Math.round(clamp(((n >> 8) & 255) * f, 0, 255));
    var b = Math.round(clamp((n & 255) * f, 0, 255));
    return 'rgb(' + r + ',' + g + ',' + b + ')';
  }

  function drawShadow(cx, cy, rx, ry, a) {
    ctx.fillStyle = 'rgba(20,40,30,' + (a || 0.18) + ')';
    ellipse(ctx, cx, cy, rx, ry);
    ctx.fill();
  }

  /** The cartoon ink line every prop in the reference carries. */
  function ink(T) {
    ctx.lineWidth = Math.max(1.2, T * 0.032);
    ctx.strokeStyle = PAL.ink;
    ctx.lineJoin = 'round';
  }

  function offView(cx) { return cx < View.cullL || cx > View.cullR; }

  function drawLaneBand(lane, row) {
    var T = View.tile;
    var y = sy(row) - T / 2;
    var x0 = -View.padX;
    var w = View.w + View.padX * 2;
    var behind = World.get(row + 1);   // the lane further up the screen

    if (lane.type === 'grass') {
      ctx.fillStyle = lane.shade ? PAL.grassA : PAL.grassB;
      ctx.fillRect(x0, y, w, T + 1);
    } else if (lane.type === 'road') {
      ctx.fillStyle = lane.shade ? PAL.road : PAL.roadB;
      ctx.fillRect(x0, y, w, T + 1);
      // Kerb shadow under the lane behind, kerb highlight at the near edge
      ctx.fillStyle = 'rgba(0,0,0,0.16)';
      ctx.fillRect(x0, y, w, T * 0.07);
      ctx.fillStyle = PAL.roadEdge;
      ctx.fillRect(x0, y + T - T * 0.05, w, T * 0.05);
      // White dashes down the middle
      ctx.save();
      ctx.strokeStyle = 'rgba(255,255,255,0.85)';
      ctx.lineWidth = Math.max(2, T * 0.05);
      ctx.lineCap = 'round';
      ctx.setLineDash([T * 0.42, T * 0.44]);
      ctx.lineDashOffset = -sx(0) % (T * 0.86);
      ctx.beginPath();
      ctx.moveTo(x0, y + T / 2);
      ctx.lineTo(x0 + w, y + T / 2);
      ctx.stroke();
      ctx.restore();
    } else if (lane.type === 'rail') {
      ctx.fillStyle = PAL.ballast;
      ctx.fillRect(x0, y, w, T + 1);
      ctx.fillStyle = 'rgba(0,0,0,0.14)';
      ctx.fillRect(x0, y, w, T * 0.07);
      ctx.fillStyle = PAL.tie;
      var step = T * 0.46;
      var startTie = x0 - mod(sx(0) - x0, step);
      for (var tx = startTie; tx < x0 + w; tx += step) {
        rr(ctx, tx, y + T * 0.17, step * 0.56, T * 0.66, T * 0.03);
        ctx.fill();
      }
      ctx.fillStyle = PAL.steel;
      ctx.fillRect(x0, y + T * 0.27, w, Math.max(2, T * 0.075));
      ctx.fillRect(x0, y + T * 0.66, w, Math.max(2, T * 0.075));
      ctx.fillStyle = 'rgba(255,255,255,0.32)';
      ctx.fillRect(x0, y + T * 0.27, w, Math.max(1, T * 0.022));
      ctx.fillRect(x0, y + T * 0.66, w, Math.max(1, T * 0.022));
    } else if (lane.type === 'river') {
      ctx.fillStyle = PAL.water;
      ctx.fillRect(x0, y, w, T + 1);
      // Streaks of foam drifting with the current
      ctx.fillStyle = 'rgba(255,255,255,0.28)';
      var ww = T * 2.4;
      var drift = mod(Game.t * lane.speed * lane.dir * T * 0.5 + row * 37, ww);
      for (var wx = x0 - ww + drift; wx < x0 + w + ww; wx += ww) {
        rr(ctx, wx, y + T * 0.30, ww * 0.34, T * 0.055, T * 0.03); ctx.fill();
        rr(ctx, wx + ww * 0.48, y + T * 0.64, ww * 0.20, T * 0.05, T * 0.03); ctx.fill();
      }
      // The lane behind is a raised slab: its earthen face drops into the water.
      if (behind.type !== 'river') {
        ctx.fillStyle = PAL.bank;
        ctx.fillRect(x0, y, w, T * 0.16);
        ctx.fillStyle = PAL.bankDark;
        ctx.fillRect(x0, y + T * 0.16, w, T * 0.04);
        ctx.fillStyle = PAL.waterDeep;
        ctx.fillRect(x0, y + T * 0.20, w, T * 0.10);
      } else {
        ctx.fillStyle = PAL.waterDeep;
        ctx.fillRect(x0, y, w, T * 0.08);
      }
    }

    // Grass meeting a lower lane in front: the near edge gets a lip so lanes
    // read as stacked slabs rather than painted stripes.
    if (lane.type === 'grass') {
      var front = World.get(row - 1);
      if (front.type === 'river') {
        ctx.fillStyle = PAL.grassEdge;
        ctx.fillRect(x0, y + T - T * 0.06, w, T * 0.06);
      } else if (front.type !== 'grass') {
        ctx.fillStyle = 'rgba(0,0,0,0.10)';
        ctx.fillRect(x0, y + T - T * 0.03, w, T * 0.03);
      }
      if (behind.type !== 'grass' && behind.type !== 'river') {
        ctx.fillStyle = 'rgba(0,0,0,0.10)';
        ctx.fillRect(x0, y, w, T * 0.05);
      }
    }

    // Cap the playable strip with hedge on both sides. The tilt widens the
    // visible band well past the screen, so these fills run into the padding.
    var edgeL = sx(CFG.MIN_X - 2.5);
    var edgeR = sx(CFG.MAX_X + 2.5);
    if (edgeL > x0) {
      ctx.fillStyle = PAL.bush;
      ctx.fillRect(x0, y, edgeL - x0, T + 1);
      ctx.fillStyle = 'rgba(0,0,0,0.16)';
      ctx.fillRect(edgeL - T * 0.06, y, T * 0.06, T + 1);
    }
    if (edgeR < x0 + w) {
      ctx.fillStyle = PAL.bush;
      ctx.fillRect(edgeR, y, x0 + w - edgeR, T + 1);
      ctx.fillStyle = 'rgba(0,0,0,0.16)';
      ctx.fillRect(edgeR, y, T * 0.06, T + 1);
    }
  }

  function drawEdgeBushes(row) {
    var T = View.tile;
    var y = sy(row);
    [CFG.MIN_X - 1.6, CFG.MAX_X + 1.6].forEach(function (x) {
      var cx = sx(x);
      if (offView(cx)) return;
      drawShadow(cx, y + T * 0.28, T * 0.38, T * 0.15, 0.18);
      ctx.fillStyle = PAL.bushDark;
      rr(ctx, cx - T * 0.40, y - T * 0.24, T * 0.80, T * 0.56, T * 0.26); ctx.fill();
      ctx.fillStyle = PAL.bushHi;
      rr(ctx, cx - T * 0.34, y - T * 0.30, T * 0.68, T * 0.40, T * 0.20); ctx.fill();
    });
  }

  /** Trees and rocks share the tile-blocking role, so they share a slot. */
  function drawBlocker(x, row, info) {
    var T = View.tile;
    var cx = sx(x), cy = sy(row);
    if (offView(cx)) return;
    if (info.rock) drawRock(cx, cy, T, info.s);
    else drawTree(cx, cy, T, info.s);
  }

  /** Rounded-cube canopy on a stubby trunk â€” the reference's low-poly tree. */
  function drawTree(cx, cy, T, s) {
    var w = T * 0.74 * s, h = T * 0.66 * s;
    var top = cy - T * 0.30 - h * 0.72;
    drawShadow(cx + T * 0.06, cy + T * 0.27, w * 0.52, T * 0.12, 0.20);

    ink(T);
    ctx.fillStyle = PAL.trunk;
    rr(ctx, cx - T * 0.075, cy - T * 0.04, T * 0.15, T * 0.36, T * 0.04);
    ctx.fill(); ctx.stroke();

    ctx.fillStyle = PAL.treeDark;
    rr(ctx, cx - w / 2, top, w, h * 1.22, w * 0.24); ctx.fill();
    ctx.fillStyle = PAL.tree;
    rr(ctx, cx - w / 2, top, w, h * 0.80, w * 0.24); ctx.fill();
    ctx.fillStyle = PAL.treeHi;
    rr(ctx, cx - w * 0.36, top + h * 0.08, w * 0.50, h * 0.22, w * 0.12); ctx.fill();
    rr(ctx, cx - w / 2, top, w, h * 1.22, w * 0.24); ctx.stroke();
  }

  function drawRock(cx, cy, T, s) {
    var w = T * 0.66 * s, h = T * 0.50 * s;
    var top = cy - T * 0.06 - h * 0.62;
    drawShadow(cx + T * 0.05, cy + T * 0.24, w * 0.50, T * 0.11, 0.20);
    ink(T);
    ctx.fillStyle = PAL.rockDark;
    rr(ctx, cx - w / 2, top, w, h * 1.12, w * 0.30); ctx.fill();
    ctx.fillStyle = PAL.rock;
    rr(ctx, cx - w / 2, top, w, h * 0.74, w * 0.30); ctx.fill();
    ctx.fillStyle = PAL.rockHi;
    rr(ctx, cx - w * 0.32, top + h * 0.10, w * 0.42, h * 0.20, w * 0.10); ctx.fill();
    rr(ctx, cx - w / 2, top, w, h * 1.12, w * 0.30); ctx.stroke();
  }

  /**
   * Lily pads are the stepping-stone alternative to logs: anchored to the
   * riverbed, and solid enough to stand on. They occupy whole columns so a hop
   * always lands cleanly on one.
   */
  function drawLilyPad(x, row, pad) {
    var T = View.tile;
    var cx = sx(x), cy = sy(row);
    if (offView(cx)) return;
    var r = T * 0.40 * pad.r;

    ink(T);
    ctx.fillStyle = PAL.padDark;
    ellipse(ctx, cx, cy + T * 0.04, r, r * 0.86); ctx.fill();
    ctx.fillStyle = PAL.pad;
    ellipse(ctx, cx, cy, r, r * 0.86); ctx.fill();
    ctx.fillStyle = PAL.padHi;
    ellipse(ctx, cx - r * 0.26, cy - r * 0.28, r * 0.36, r * 0.20); ctx.fill();

    // The signature wedge notch, cut back to water.
    ctx.fillStyle = PAL.water;
    ctx.beginPath();
    ctx.moveTo(cx, cy);
    ctx.arc(cx, cy, r * 1.08, pad.a - 0.33, pad.a + 0.33);
    ctx.closePath();
    ctx.fill();

    // Ink the pad rim (minus the notch) so it sits in the water like the props do
    ctx.beginPath();
    ctx.arc(cx, cy, r, pad.a + 0.33, pad.a - 0.33 + Math.PI * 2);
    ctx.stroke();

    if (pad.flower) {
      ctx.fillStyle = PAL.flower;
      for (var i = 0; i < 5; i++) {
        var a = (i / 5) * Math.PI * 2;
        ellipse(ctx, cx + Math.cos(a) * r * 0.24, cy + Math.sin(a) * r * 0.22, r * 0.17, r * 0.15);
        ctx.fill();
      }
      ctx.fillStyle = PAL.flowerMid;
      ellipse(ctx, cx, cy, r * 0.15, r * 0.13); ctx.fill();
    }
  }

  /**
   * Vehicles: a rounded chassis with a separate raised cabin, wheels with rims,
   * and an ink outline â€” you read a top face and a side face like the reference.
   */
  function drawVehicle(lane, cx, row) {
    var T = View.tile;
    var len = lane.len * T;
    var x = cx - len / 2;
    if (x > View.cullR || x + len < View.cullL) return;

    var y = sy(row);
    var flip = lane.dir < 0;
    var base = lane.itemColor;
    var side = shade(base, 0.74);
    var roofC = shade(base, 1.14);
    var isBus = lane.vType === 'bus', isTruck = lane.vType === 'truck';

    var bodyH = T * (isBus ? 0.56 : isTruck ? 0.54 : 0.50);
    var lift = T * 0.22;
    var bodyTop = y - bodyH / 2 - lift * 0.3;

    drawShadow(cx + T * 0.03, y + T * 0.28, len * 0.47, T * 0.15, 0.24);
    ink(T);

    // Wheels: dark tyre, light rim
    var wy = bodyTop + bodyH * 0.74;
    var ww = len * 0.17, wh = T * 0.22;
    [0.13, 0.70].forEach(function (f) {
      ctx.fillStyle = '#23272c';
      rr(ctx, x + len * f, wy, ww, wh, T * 0.07); ctx.fill(); ctx.stroke();
      ctx.fillStyle = '#9aa3ab';
      rr(ctx, x + len * f + ww * 0.28, wy + wh * 0.30, ww * 0.44, wh * 0.45, T * 0.04); ctx.fill();
    });

    // Chassis: shaded side face, then lit top
    ctx.fillStyle = side;
    rr(ctx, x, bodyTop + bodyH * 0.32, len, bodyH * 0.80, T * 0.15); ctx.fill();
    ctx.fillStyle = base;
    rr(ctx, x, bodyTop, len, bodyH * 0.90, T * 0.15); ctx.fill();
    ctx.fillStyle = 'rgba(255,255,255,0.16)';
    rr(ctx, x + len * 0.05, bodyTop + bodyH * 0.06, len * 0.90, bodyH * 0.16, T * 0.06); ctx.fill();
    rr(ctx, x, bodyTop, len, bodyH * 1.12, T * 0.15); ctx.stroke();

    if (isTruck) {
      var boxW = len * 0.62, cabW = len * 0.36;
      var boxX = flip ? x + len - boxW : x;
      var cabX = flip ? x : x + len - cabW;

      ctx.fillStyle = shade(PAL.cargo, 0.80);
      rr(ctx, boxX, bodyTop - bodyH * 0.46, boxW, bodyH * 1.24, T * 0.10); ctx.fill();
      ctx.fillStyle = PAL.cargo;
      rr(ctx, boxX, bodyTop - bodyH * 0.68, boxW, bodyH * 1.02, T * 0.10); ctx.fill();
      ctx.fillStyle = 'rgba(255,255,255,0.5)';
      rr(ctx, boxX + T * 0.05, bodyTop - bodyH * 0.60, boxW - T * 0.10, bodyH * 0.22, T * 0.06); ctx.fill();
      rr(ctx, boxX, bodyTop - bodyH * 0.68, boxW, bodyH * 1.46, T * 0.10); ctx.stroke();

      ctx.fillStyle = side;
      rr(ctx, cabX, bodyTop - bodyH * 0.12, cabW, bodyH * 0.98, T * 0.11); ctx.fill();
      ctx.fillStyle = base;
      rr(ctx, cabX, bodyTop - bodyH * 0.32, cabW, bodyH * 0.90, T * 0.11); ctx.fill();
      rr(ctx, cabX, bodyTop - bodyH * 0.32, cabW, bodyH * 1.18, T * 0.11); ctx.stroke();
      ctx.fillStyle = PAL.glass;
      rr(ctx, cabX + cabW * (flip ? 0.10 : 0.26), bodyTop - bodyH * 0.20,
         cabW * 0.62, bodyH * 0.46, T * 0.06);
      ctx.fill(); ctx.stroke();
    } else {
      var ghW = len * (isBus ? 0.90 : 0.60);
      var ghX = x + (len - ghW) / 2 + (flip ? -len * 0.05 : len * 0.05);
      var ghH = bodyH * 0.84;
      var ghTop = bodyTop - ghH * 0.66;

      ctx.fillStyle = shade(base, 0.90);
      rr(ctx, ghX, ghTop + ghH * 0.24, ghW, ghH, T * 0.12); ctx.fill();
      ctx.fillStyle = roofC;
      rr(ctx, ghX, ghTop, ghW, ghH * 0.92, T * 0.12); ctx.fill();
      rr(ctx, ghX, ghTop, ghW, ghH * 1.24, T * 0.12); ctx.stroke();

      ctx.fillStyle = PAL.glass;
      if (isBus) {
        for (var i = 0; i < 4; i++) {
          rr(ctx, ghX + ghW * (0.06 + i * 0.235), ghTop + ghH * 0.22,
             ghW * 0.18, ghH * 0.44, T * 0.045);
          ctx.fill(); ctx.stroke();
        }
      } else {
        rr(ctx, ghX + ghW * 0.09, ghTop + ghH * 0.20, ghW * 0.82, ghH * 0.48, T * 0.06);
        ctx.fill(); ctx.stroke();
        ctx.fillStyle = 'rgba(255,255,255,0.42)';
        rr(ctx, ghX + ghW * 0.12, ghTop + ghH * 0.24, ghW * 0.30, ghH * 0.36, T * 0.05);
        ctx.fill();
      }
    }

    // Lights
    ctx.fillStyle = '#fff6cf';
    rr(ctx, flip ? x + len * 0.015 : x + len * 0.955, bodyTop + bodyH * 0.36,
       len * 0.03, bodyH * 0.26, T * 0.03);
    ctx.fill();
    ctx.fillStyle = '#ff6b5c';
    rr(ctx, flip ? x + len * 0.955 : x + len * 0.015, bodyTop + bodyH * 0.36,
       len * 0.03, bodyH * 0.24, T * 0.03);
    ctx.fill();
  }

  function drawLog(cx, row, len) {
    var T = View.tile;
    var w = len * T;
    var x = cx - w / 2;
    if (x > View.cullR || x + w < View.cullL) return;
    var y = sy(row);
    var h = T * 0.66;
    var top = y - h / 2 - T * 0.06;

    // Reflection under the log
    ctx.fillStyle = 'rgba(0,40,80,0.22)';
    rr(ctx, x + T * 0.03, top + h * 0.30, w, h * 0.92, h / 2); ctx.fill();

    ink(T);
    ctx.fillStyle = PAL.logDark;
    rr(ctx, x, top + h * 0.20, w, h * 0.86, h / 2); ctx.fill();
    ctx.fillStyle = PAL.log;
    rr(ctx, x, top, w, h * 0.84, h / 2); ctx.fill();
    ctx.fillStyle = PAL.logHi;
    rr(ctx, x + T * 0.16, top + h * 0.12, w - T * 0.32, h * 0.18, h * 0.09); ctx.fill();
    rr(ctx, x, top, w, h * 1.06, h / 2); ctx.stroke();

    // End grain on the near-right end
    var ex = x + w - h * 0.30, ey = top + h * 0.52;
    ctx.fillStyle = PAL.logRing;
    ellipse(ctx, ex, ey, h * 0.24, h * 0.42); ctx.fill(); ctx.stroke();
    ctx.fillStyle = PAL.logCore;
    ellipse(ctx, ex, ey, h * 0.13, h * 0.24); ctx.fill();
    ctx.fillStyle = PAL.logRing;
    ellipse(ctx, ex, ey, h * 0.05, h * 0.09); ctx.fill();
  }

  function drawTrain(lane, row, st) {
    var T = View.tile;
    var a = Math.min(st.head, st.tail), b = Math.max(st.head, st.tail);
    var x = sx(a), w = (b - a) * T;
    if (x > View.cullR || x + w < View.cullL) return;
    var y = sy(row);
    var h = T * 0.74;
    var lift = T * 0.16;

    drawShadow((x + x + w) / 2, y + T * 0.26, w * 0.48, T * 0.13, 0.26);
    ink(T);

    ctx.fillStyle = shade(PAL.trainBody, 0.70);
    rr(ctx, x, y - h / 2 - lift + h * 0.30, w, h * 0.82, T * 0.12); ctx.fill();
    ctx.fillStyle = PAL.trainBody;
    rr(ctx, x, y - h / 2 - lift, w, h * 0.88, T * 0.12); ctx.fill();
    ctx.fillStyle = 'rgba(255,255,255,0.20)';
    rr(ctx, x + T * 0.1, y - h / 2 - lift + h * 0.08, w - T * 0.2, h * 0.18, T * 0.07); ctx.fill();
    rr(ctx, x, y - h / 2 - lift, w, h * 1.12, T * 0.12); ctx.stroke();

    ctx.fillStyle = PAL.trainNose;
    var nose = lane.dir > 0 ? x + w - T * 0.85 : x;
    rr(ctx, nose, y - h / 2 - lift, T * 0.85, h * 0.88, T * 0.12); ctx.fill(); ctx.stroke();

    ctx.fillStyle = PAL.glass;
    var step = T * 1.05;
    for (var wx = x + T * 1.0; wx < x + w - T * 0.9; wx += step) {
      rr(ctx, wx, y - h * 0.26 - lift, T * 0.46, h * 0.28, T * 0.05);
      ctx.fill(); ctx.stroke();
    }

    ctx.fillStyle = 'rgba(255,255,255,0.16)';
    for (var i = 0; i < 5; i++) {
      var sxp = lane.dir > 0 ? x - T * (0.6 + i * 0.9) : x + w + T * (0.6 + i * 0.9);
      ctx.fillRect(sxp, y - h * 0.3 + i * T * 0.13, T * 0.7, Math.max(1, T * 0.045));
    }
  }

  /**
   * Level-crossing signals. Pinned to the screen edge rather than to a world
   * column: the playfield is far wider than a phone viewport, so a
   * world-anchored signal would sit off-screen and the 1.2s warning â€” the whole
   * point of it â€” would never be seen. Drawn after the tilt, rotated by hand so
   * they still sit square to the track.
   */
  function drawCrossing(row, st) {
    var T = View.tile;
    var lit = st.warn && st.blink;
    var inset = Math.max(T * 0.6, 24);

    [View.w - inset].forEach(function (screenX) {
      var p = laneEdgePoint(row, screenX);
      if (p.y < -T * 2 || p.y > View.h + T * 2) return;

      ctx.save();
      ctx.translate(p.x, p.y);
      ctx.rotate(CFG.WORLD_ANGLE);
      ink(T);

      ctx.fillStyle = '#8b9299';
      rr(ctx, -T * 0.05, -T * 0.10, T * 0.10, T * 0.66, T * 0.03); ctx.fill(); ctx.stroke();

      ctx.save();
      ctx.translate(0, -T * 0.52);
      [-0.72, 0.72].forEach(function (rot) {
        ctx.save();
        ctx.rotate(rot);
        ctx.fillStyle = '#f4f1e6';
        rr(ctx, -T * 0.26, -T * 0.045, T * 0.52, T * 0.09, T * 0.04); ctx.fill(); ctx.stroke();
        ctx.restore();
      });
      ctx.restore();

      [-1, 1].forEach(function (dx, i) {
        var on = lit && ((i === 0) === (st.blink));
        var lx = dx * T * 0.15, ly = -T * 0.16;
        if (on) {
          ctx.fillStyle = 'rgba(255,60,60,0.34)';
          ellipse(ctx, lx, ly, T * 0.26, T * 0.26); ctx.fill();
        }
        ctx.fillStyle = on ? PAL.warnOn : PAL.warnOff;
        ellipse(ctx, lx, ly, T * 0.11, T * 0.11); ctx.fill(); ctx.stroke();
        if (on) {
          ctx.fillStyle = 'rgba(255,255,255,0.75)';
          ellipse(ctx, lx - T * 0.035, ly - T * 0.035, T * 0.04, T * 0.04); ctx.fill();
        }
      });

      ctx.restore();
    });
  }

  var DIR_ANGLE = [0, Math.PI / 2, Math.PI, -Math.PI / 2];

  function drawChicken(cx, cy, size, dirIdx, opts) {
    opts = opts || {};
    var alpha = opts.alpha == null ? 1 : opts.alpha;
    var body = opts.body || '#ffffff';
    var sxs = opts.sx == null ? 1 : opts.sx;
    var sys = opts.sy == null ? 1 : opts.sy;
    var T = View.tile;

    ctx.save();
    ctx.globalAlpha *= alpha;
    ctx.translate(cx, cy);
    ctx.scale(sxs, sys);
    ctx.rotate(DIR_ANGLE[dirIdx] || 0);
    ink(T);

    var s = size;

    // Feet, peeking out below the body
    ctx.fillStyle = '#f39c2b';
    rr(ctx, -s * 0.19, s * 0.34, s * 0.14, s * 0.15, s * 0.05); ctx.fill(); ctx.stroke();
    rr(ctx, s * 0.05, s * 0.34, s * 0.14, s * 0.15, s * 0.05); ctx.fill(); ctx.stroke();

    // Tail tuft (toward the viewer)
    ctx.fillStyle = shade(body === '#ffffff' ? '#e9eef3' : body, 0.96);
    ctx.beginPath();
    ctx.moveTo(-s * 0.10, s * 0.28);
    ctx.lineTo(s * 0.10, s * 0.28);
    ctx.lineTo(0, s * 0.46);
    ctx.closePath();
    ctx.fill(); ctx.stroke();

    // Body
    ctx.fillStyle = body;
    ellipse(ctx, 0, s * 0.08, s * 0.37, s * 0.40); ctx.fill(); ctx.stroke();

    // Folded-wing crease down the back
    ctx.beginPath();
    ctx.moveTo(-s * 0.20, s * 0.02);
    ctx.quadraticCurveTo(-s * 0.06, s * 0.30, 0, s * 0.12);
    ctx.quadraticCurveTo(s * 0.06, s * 0.30, s * 0.20, s * 0.02);
    ctx.stroke();

    // Head
    ctx.fillStyle = body;
    ellipse(ctx, 0, -s * 0.28, s * 0.26, s * 0.25); ctx.fill(); ctx.stroke();

    // Beak tip peeking past the head
    ctx.fillStyle = '#f5a623';
    ctx.beginPath();
    ctx.moveTo(-s * 0.07, -s * 0.44);
    ctx.lineTo(s * 0.07, -s * 0.44);
    ctx.lineTo(0, -s * 0.56);
    ctx.closePath();
    ctx.fill(); ctx.stroke();

    // Comb
    ctx.fillStyle = '#e8453c';
    ellipse(ctx, -s * 0.09, -s * 0.46, s * 0.075, s * 0.07); ctx.fill();
    ellipse(ctx, s * 0.01, -s * 0.51, s * 0.08, s * 0.075); ctx.fill();
    ellipse(ctx, s * 0.10, -s * 0.46, s * 0.07, s * 0.065); ctx.fill();
    ctx.beginPath();
    ctx.arc(-s * 0.09, -s * 0.46, s * 0.072, Math.PI, 0);
    ctx.arc(s * 0.01, -s * 0.51, s * 0.078, Math.PI, 0);
    ctx.arc(s * 0.10, -s * 0.46, s * 0.068, Math.PI, 0);
    ctx.stroke();

    ctx.restore();
  }

  function drawPlayer() {
    var T = View.tile;
    var cx = sx(Player.fx);
    var groundY = sy(Player.frow);
    var size = T * 0.86;

    if (Player.death) { drawDeath(cx, groundY, size); return; }

    var hop = Player.hopping ? Math.sin(Math.PI * Player.hopT) : 0;
    var lift = hop * CFG.HOP_ARC * T;
    var stretch = Player.hopping ? (1 + hop * 0.16 - (1 - hop) * 0.10) : (1 - Player.squash * 0.26);
    var widen = Player.hopping ? (1 - hop * 0.10 + (1 - hop) * 0.08) : (1 + Player.squash * 0.24);

    drawShadow(cx, groundY + T * 0.26, T * 0.30 * (1 - hop * 0.3), T * 0.13 * (1 - hop * 0.3), 0.24);

    if (Player.status === 'banked') {
      ctx.save();
      ctx.globalAlpha = 0.45 + Math.sin(Game.t * 5) * 0.16;
      ctx.fillStyle = 'rgba(126,200,80,0.35)';
      ellipse(ctx, cx, groundY - T * 0.1, T * 0.58, T * 0.58); ctx.fill();
      ctx.restore();
    }

    drawChicken(cx, groundY - lift - T * 0.06, size, Player.dir, { sx: widen, sy: stretch });
  }

  function drawDeath(cx, cy, size) {
    var T = View.tile;
    var d = Player.death;
    var p = Math.min(1, d.t / 0.42);

    if (d.kind === 'splash') {
      var sink = easeOutCubic(p);
      ctx.save();
      ctx.globalAlpha = 1 - sink * 0.85;
      drawChicken(cx, cy - T * 0.05 + sink * T * 0.22, size * (1 - sink * 0.4), Player.dir, {});
      ctx.restore();
      ctx.strokeStyle = 'rgba(255,255,255,' + (0.75 * (1 - p)) + ')';
      ctx.lineWidth = Math.max(1.5, T * 0.06);
      for (var i = 0; i < 3; i++) {
        var rp = clamp(p * 1.6 - i * 0.22, 0, 1);
        if (rp <= 0) continue;
        ellipse(ctx, cx, cy + T * 0.04, T * (0.16 + rp * 0.62), T * (0.06 + rp * 0.26));
        ctx.stroke();
      }
    } else if (d.kind === 'train') {
      var fly = easeOutCubic(p);
      var off = d.dir * fly * T * 5.5;
      ctx.save();
      ctx.globalAlpha = 1 - fly * 0.6;
      ctx.translate(cx + off, cy - T * 0.05 - Math.sin(p * Math.PI) * T * 1.4);
      ctx.rotate(fly * d.dir * 7);
      drawChicken(0, 0, size, Player.dir, { sy: 0.55, sx: 1.3 });
      ctx.restore();
    } else {
      var f = easeOutCubic(p);
      drawShadow(cx, cy + T * 0.24, T * 0.34, T * 0.14, 0.3);
      drawChicken(cx, cy + f * T * 0.20 - T * 0.05, size, Player.dir, {
        sx: 1 + f * 0.75, sy: 1 - f * 0.80
      });
      ctx.fillStyle = 'rgba(255,255,255,' + (0.85 * (1 - p)) + ')';
      for (var k = 0; k < 5; k++) {
        var a = (k / 5) * Math.PI * 2 + 0.5;
        ellipse(ctx, cx + Math.cos(a) * T * p * 1.1, cy + Math.sin(a) * T * p * 0.6 - T * 0.2,
                T * 0.07, T * 0.10);
        ctx.fill();
      }
    }
  }

  function render() {
    var T = View.tile;

    ctx.fillStyle = PAL.void;
    ctx.fillRect(0, 0, View.w, View.h);

    ctx.save();
    if (Game.shake > 0) {
      ctx.translate((Math.random() - 0.5) * Game.shake * T * 0.3,
                    (Math.random() - 0.5) * Game.shake * T * 0.3);
    }
    applyWorldTilt();

    var extra = Math.ceil(View.padY / T) + 2;
    var rowsUp = Math.ceil((View.h * CFG.PLAYER_SCREEN_Y) / T) + extra;
    var rowsDown = Math.ceil((View.h * (1 - CFG.PLAYER_SCREEN_Y)) / T) + extra;
    var topRow = Math.ceil(Game.cam.row) + rowsUp;
    var botRow = Math.floor(Game.cam.row) - rowsDown;
    World.ensure(topRow + 4);

    // Pass 1 â€” lane surfaces, far to near.
    for (var r = topRow; r >= botRow; r--) {
      if (r < -PAD) continue;
      drawLaneBand(World.get(r), r);
    }

    // Pass 2 â€” props and actors, far to near (painter's order), clipped to the
    // playable strip so traffic spawning off-world never draws on the hedge.
    var playerBucket = Math.round(Player.frow);
    var ghostBucket = Game.ghost.active ? Math.round(Game.ghost.frow) : null;
    var crossings = [];

    ctx.save();
    var clipL = sx(CFG.MIN_X - 2.5);
    var clipR = sx(CFG.MAX_X + 2.5);
    ctx.beginPath();
    ctx.rect(clipL, -View.padY - View.h, Math.max(0, clipR - clipL), View.h * 3 + View.padY * 2);
    ctx.clip();

    for (var row = topRow; row >= botRow; row--) {
      if (row < -PAD) continue;
      var lane = World.get(row);
      drawEdgeBushes(row);

      if (lane.type === 'grass') {
        for (var key in lane.trees) drawBlocker(+key, row, lane.trees[key]);
      } else if (lane.type === 'road') {
        for (var i = 0; i < lane.items.length; i++) {
          lane.itemColor = lane.items[i].color;
          drawVehicle(lane, sx(itemX(lane, i, Game.t)), row);
        }
      } else if (lane.type === 'river') {
        if (lane.pads) {
          for (var q in lane.pads) drawLilyPad(+q, row, lane.pads[q]);
        }
        for (var j = 0; j < lane.items.length; j++) {
          drawLog(sx(itemX(lane, j, Game.t)), row, lane.items[j].len);
        }
      } else if (lane.type === 'rail') {
        var st = trainState(lane, Game.t);
        crossings.push({ row: row, st: st });
        if (st.active) drawTrain(lane, row, st);
      }

      if (ghostBucket === row) drawGhost();
      if (playerBucket === row && Game.phase !== 'menu') drawPlayer();
    }

    if (ghostBucket === null || ghostBucket > topRow || ghostBucket < botRow) drawGhost();
    if (Game.phase !== 'menu' && (playerBucket > topRow || playerBucket < botRow)) drawPlayer();

    ctx.restore(); // release the playfield clip
    ctx.restore(); // release the world tilt

    // Screen-space layer: warning wash, signals, labels, popups.
    for (var wi = 0; wi < crossings.length; wi++) {
      drawCrossingWash(crossings[wi].row, crossings[wi].st);
    }
    // Show a post only where it carries information: any track that is warning
    // or occupied, plus the next track ahead.
    var nearest = null;
    for (var wj = 0; wj < crossings.length; wj++) {
      var cr = crossings[wj];
      if (cr.row < Player.frow) continue;
      if (nearest === null || cr.row < nearest.row) nearest = cr;
    }
    for (var wk = 0; wk < crossings.length; wk++) {
      var c2 = crossings[wk];
      if (c2.st.warn || c2.st.active || c2 === nearest) drawCrossing(c2.row, c2.st);
    }

    if (Game.phase !== 'menu') Popups.draw(ctx, View);
    drawGhostOverlay();
  }

  /** Red wash across a track that is about to be hit by a train. */
  function drawCrossingWash(row, st) {
    if (!st.warn || !st.blink) return;
    var T = View.tile;
    var a = laneEdgePoint(row, -View.w);
    var b = laneEdgePoint(row, View.w * 2);
    ctx.save();
    ctx.strokeStyle = 'rgba(255,45,45,0.16)';
    ctx.lineWidth = T * CFG.WORLD_SQUASH;
    ctx.beginPath();
    ctx.moveTo(a.x, a.y);
    ctx.lineTo(b.x, b.y);
    ctx.stroke();
    ctx.restore();
  }

  /* Movement + collision                                                    */
  /* ====================================================================== */

  function currentLane() { return World.get(Math.round(Player.row)); }

  function tryMove(dx, dy) {
    if (Game.phase !== 'playing' || !Player.isRunning()) return;

    if (Player.hopping) {
      // Buffer one input so fast tapping feels responsive rather than dropped.
      if (Player.hopT > 0.45) Player.queued = { dx: dx, dy: dy };
      return;
    }

    Player.dir = dy > 0 ? 0 : dy < 0 ? 2 : (dx > 0 ? 1 : 3);

    var lane = currentLane();
    var onDriftingLog = lane.type === 'river' && lane.mode === 'logs' && Player.onLog;
    var baseX = onDriftingLog ? Player.x : Math.round(Player.x);
    var targetRow = Math.round(Player.row) + dy;
    var targetX = baseX + dx;

    targetRow = Math.max(-2, targetRow);
    var destLane = World.get(targetRow);
    // Only a drifting-log lane keeps a fractional x; pads sit on whole columns.
    if (!(destLane.type === 'river' && destLane.mode === 'logs')) targetX = Math.round(targetX);
    targetX = clamp(targetX, CFG.MIN_X, CFG.MAX_X);

    // Trees are solid.
    if (destLane.type === 'grass' && destLane.trees[Math.round(targetX)]) return;

    Player.fromX = Player.x; Player.fromRow = Player.row;
    Player.toX = targetX; Player.toRow = targetRow;
    Player.hopping = true;
    Player.hopT = 0;
    Player.onLog = false;
    SFX.hop();
  }

  function landed() {
    Player.x = Player.toX;
    Player.row = Player.toRow;
    Player.fx = Player.x;
    Player.frow = Player.row;
    Player.hopping = false;
    Player.squash = 1;

    var lane = currentLane();

    // +1 per NEW forward tile (backtracking never re-scores).
    if (Player.row > Player.maxRow) {
      Player.pending += (Player.row - Player.maxRow) * CFG.TILE_POINTS;
      Player.maxRow = Player.row;
      UI.pulsePending();
    }

    // River landing must find a log or a lily pad.
    if (lane.type === 'river') {
      var lg = platformUnder(lane, Player.x, Game.t);
      if (!lg) { onDeath('splash'); return; }
      Player.onLog = true;
      if (lg.pad) Player.x = Player.fx = lg.cx;   // settle onto the pad
    }
  }

  function collisionCheck(dt) {
    if (!Player.isRunning()) return;
    var lane = currentLane();

    if (lane.type === 'road') {
      if (Player.hopping) return;                 // brief i-frames mid-hop
      for (var i = 0; i < lane.items.length; i++) {
        var cx = itemX(lane, i, Game.t);
        var d = Math.abs(Player.x - cx);
        if (d < lane.len / 2 + 0.40) { onDeath('squash'); return; }
        // Near miss: it got close enough to ruffle feathers but didn't hit.
        var nkey = lane.row + ':' + i;
        if (d < lane.len / 2 + CFG.NEAR_MISS) {
          if (!Game.nearFlags[nkey]) {
            // A near miss is felt, not announced â€” a shove of the camera and a
            // low thrum. No text: this fires often and would read as clutter.
            Game.nearFlags[nkey] = true;
            Game.shake = Math.max(Game.shake, 0.45);
            SFX.near();
          }
        } else if (d > lane.len / 2 + CFG.NEAR_MISS * 1.8) {
          Game.nearFlags[nkey] = false;
        }
      }
    } else if (lane.type === 'rail') {
      var st = trainState(lane, Game.t);
      if (st.active) {
        var a = Math.min(st.head, st.tail), b = Math.max(st.head, st.tail);
        if (Player.x > a - 0.4 && Player.x < b + 0.4) { onDeath('train', lane.dir); return; }
      }
    } else if (lane.type === 'river' && !Player.hopping) {
      var lg = platformUnder(lane, Player.x, Game.t);
      if (!lg) { onDeath('splash'); return; }
      Player.onLog = true;
      // A log carries you at exactly its speed; a lily pad is anchored, so
      // standing on one is a genuine safe pause in the middle of the water.
      if (!lg.pad) {
        Player.x += lane.dir * lane.speed * dt;
        Player.fx = Player.x;
        if (Player.x < CFG.MIN_X - 0.9 || Player.x > CFG.MAX_X + 0.9) { onDeath('splash'); return; }
      }
    }
  }

  /* ====================================================================== */
  /* Run resolution â€” the whole game lives here                              */
  /* ====================================================================== */

  /**
   * DEATH. Everything pending is forfeit. Nothing is ever auto-saved here:
   * a dead run banks exactly 0.
   */
  function onDeath(kind, dir) {
    if (!Player.isRunning()) return;
    var lost = Player.pending;

    Player.pending = 0;
    Player.banked = 0;
    Player.status = 'dead';
    Player.hopping = false;
    Player.queued = null;
    Player.death = { kind: kind, t: 0, dir: dir || 1 };

    Game.shake = kind === 'train' ? 1.6 : kind === 'squash' ? 1.2 : 0.5;
    Game.phase = 'resolved';
    Game.resolveAt = Game.t;

    Popups.add(lost > 0 ? 'WIPED OUT  â€”  ' + lost + ' LOST' : 'WIPED OUT',
               PAL.crimson, { size: 0.40, vy: 1.0, centre: true, life: 1.3 });
    SFX.death();
    if (navigator.vibrate) { try { navigator.vibrate(kind === 'splash' ? 40 : [0, 60, 40, 90]); } catch (e) {} }

    Net.sendState(true);
    Net.emit('died', {});
    UI.refreshCashout();
    if (Game.solo) setTimeout(function () { UI.showEndSolo(); }, 1300);
  }

  /**
   * CASHOUT. Locks the pending score into banked, ends the run safely, and
   * tells the rival what they now have to beat.
   */
  function bankScore() {
    if (Game.phase !== 'playing' || !Player.isRunning()) return;
    if (Player.pending <= 0) return;

    Player.banked = Player.pending;
    Player.status = 'banked';

    if (Game.solo) setTimeout(function () { UI.showEndSolo(); }, 1400);

  function update(dt) {
    if (Game.phase === 'countdown' || Game.phase === 'playing' || Game.phase === 'resolved') {
      Game.t = Math.max(0, (Net.now() - Game.startAt) / 1000);
    }

    if (Game.phase === 'countdown' && Net.now() >= Game.startAt) {
      Game.phase = 'playing';
      UI.hideCountdown();
      UI.refreshCashout();
    }

    if (Player.death) Player.death.t += dt;
    if (Player.squash > 0) Player.squash = Math.max(0, Player.squash - dt * 7);
    if (Game.shake > 0) Game.shake = Math.max(0, Game.shake - dt * 4.5);
    Popups.update(dt);

    if (Player.hopping) {
      Player.hopT += dt * (1000 / CFG.HOP_MS);
      if (Player.hopT >= 1) {
        Player.hopT = 1;
        landed();
        if (Player.queued && Player.isRunning()) {
          var q = Player.queued; Player.queued = null;
          tryMove(q.dx, q.dy);
        }
      } else {
        var e = Player.hopT;
        Player.fx = lerp(Player.fromX, Player.toX, e);
        Player.frow = lerp(Player.fromRow, Player.toRow, e);
      }
    } else {
      Player.fx = Player.x;
      Player.frow = Player.row;
    }

    if (Game.phase === 'playing') collisionCheck(dt);

    // Train proximity callout â€” one per approaching train.
    if (Game.phase === 'playing' && Player.isRunning()) {
      for (var r = Math.round(Player.row) + 1; r <= Math.round(Player.row) + 3; r++) {
        var ln = World.get(r);
        if (ln.type !== 'rail') continue;
        var st = trainState(ln, Game.t);
        var key = r + ':' + Math.floor((Game.t + ln.phase) / ln.period);
        if (st.warn && !Game.trainAlerted[key]) {
          Game.trainAlerted[key] = true;
          Popups.add('TRAIN', PAL.crimson, { size: 0.30, vy: 1.1, centre: true, life: 0.8 });
          Game.shake = Math.max(Game.shake, 0.2);
        }
      }
    }

    // Camera â€” smooth follow with a little look-ahead. Once your run is
    // resolved the camera drifts to the rival so you can sweat their attempt.
    if (Game.phase !== 'menu') {
      var followRow = Player.frow, followX = Player.fx;
      if (!Game.solo && Game.phase === 'resolved' &&
          Game.t - Game.resolveAt > 1.1 &&
          Game.ghost.active && Game.ghost.status === 'running') {
        followRow = Game.ghost.frow;
        followX = Game.ghost.fx;
        if (!Game.spectating) {
          Game.spectating = true;
          UI.toast(Player.status === 'banked' ? 'BANKED â€” WATCHING RIVAL' : 'WATCHING RIVAL');
        }
      }
      var targetRow = followRow + 0.35;
      var camReach = Math.max(0, CFG.MAX_X - (View.w / 2) / View.tile + 0.9);
      var targetX = clamp(followX, -camReach, camReach);
      var k = 1 - Math.pow(0.0016, dt);
      Game.cam.row = lerp(Game.cam.row, targetRow, k);
      Game.cam.x = lerp(Game.cam.x, targetX, k);
    }

    // Ghost interpolation.
    var g = Game.ghost;
    if (g.active) {
      var gk = 1 - Math.pow(0.0009, dt);
      g.fx = lerp(g.fx, g.x, gk);
      g.frow = lerp(g.frow, g.row, gk);
    }

    UI.tick();
  }

  var rafId = 0;
  function loop(now) {
    rafId = requestAnimationFrame(loop);
    var dt = Game.lastFrame ? Math.min(0.05, (now - Game.lastFrame) / 1000) : 0.016;
    Game.lastFrame = now;
    update(dt);
    render();
  }

  var Touch_ = { x: 0, y: 0, t: 0, id: null };
  var SWIPE_PX = 26;

  function onStart(x, y, id) {
    Touch_.x = x; Touch_.y = y; Touch_.t = performance.now(); Touch_.id = id;
    SFX.ensure(); // first gesture unlocks WebAudio on iOS
  }

  function onEnd(x, y) {
    var dx = x - Touch_.x;
    var dy = y - Touch_.y;
    var adx = Math.abs(dx), ady = Math.abs(dy);
    var dtms = performance.now() - Touch_.t;

    if (adx < SWIPE_PX && ady < SWIPE_PX) {
      if (dtms < 700) tryMove(0, 1);              // tap = hop forward
      return;
    }
    if (adx > ady) tryMove(dx > 0 ? 1 : -1, 0);   // swipe left / right
    else if (dy > 0) tryMove(0, -1);              // swipe down = back
    else tryMove(0, 1);                           // swipe up = forward
  }

  canvas.addEventListener('touchstart', function (e) {
    e.preventDefault();
    var t = e.changedTouches[0];
    if (Touch_.id !== null) return;
    onStart(t.clientX, t.clientY, t.identifier);
  }, { passive: false });

  canvas.addEventListener('touchmove', function (e) { e.preventDefault(); }, { passive: false });

  canvas.addEventListener('touchend', function (e) {
    e.preventDefault();
    for (var i = 0; i < e.changedTouches.length; i++) {
      var t = e.changedTouches[i];
      if (t.identifier === Touch_.id) { Touch_.id = null; onEnd(t.clientX, t.clientY); }
    }
  }, { passive: false });

  canvas.addEventListener('touchcancel', function () { Touch_.id = null; }, { passive: false });

  canvas.addEventListener('mousedown', function (e) { onStart(e.clientX, e.clientY, 'm'); });
  canvas.addEventListener('mouseup', function (e) {
    if (Touch_.id === 'm') { Touch_.id = null; onEnd(e.clientX, e.clientY); }
  });

  var Net = {
    socket: null,
    offset: 0,
    samples: [],
    code: null,
    slot: 0,
    timer: 0,

    now: function () { return Date.now() + this.offset; },

    connect: function () {
      if (typeof io === 'undefined') { UI.setConn('OFFLINE â€” SOLO ONLY', false); return; }
      var self = this;

      // Render's free tier sleeps when idle; the first hit can take ~30s. Poke
      // the health route right away so the server is waking while the menu
      // loads, and say so instead of looking broken.
      try { fetch(SOCKET_URL + '/health', { mode: 'cors', cache: 'no-store' }).catch(function () {}); } catch (e) {}
      var slow = setTimeout(function () {
        if (!self.socket || !self.socket.connected) UI.setConn('WAKING SERVERâ€¦ (~30S)', false);
      }, 2500);

      var s = io(SOCKET_URL, {
        transports: ['websocket', 'polling'],
        reconnectionDelayMax: 4000,
        timeout: 20000
      });
      this.socket = s;

      s.on('connect', function () { clearTimeout(slow); UI.setConn('ONLINE', true); self.syncClock(); });
      s.on('disconnect', function () { UI.setConn('RECONNECTINGâ€¦', false); });
      s.on('connect_error', function () { UI.setConn('WAKING SERVERâ€¦ RETRYING', false); });

      s.on('tsync', function (d) {
        var now = Date.now();
        var rtt = now - d.t0;
        self.samples.push(d.server + rtt / 2 - now);
        if (self.samples.length > 7) self.samples.shift();
        var sorted = self.samples.slice().sort(function (a, b) { return a - b; });
        self.offset = sorted[(sorted.length / 2) | 0];
      });

      s.on('room', function (room) { Net.code = room.code; UI.setLobbyCode(room.code); });
      s.on('matchStart', function (d) { Net.slot = d.you.slot; UI.beginMatch(d); });

      s.on('oppState', function (d) {
        var g = Game.ghost;
        if (!g.active) { g.fx = d.x; g.frow = d.y; g.active = true; }
        g.x = d.x; g.row = d.y; g.pending = d.p;
      });

      s.on('oppDied', function () {
        var g = Game.ghost;
        g.status = 'dead';
        g.pending = 0;
        g.banked = 0;
        UI.toast('RIVAL WIPED OUT â€” THEY BANKED 0');
        Popups.add('RIVAL WIPED OUT', PAL.neon, { size: 0.34, vy: 1.0, centre: true, life: 1.2 });
      });

      s.on('oppCashout', function (d) {
        var g = Game.ghost;
        g.status = 'banked';
        g.banked = d.banked;
        UI.toast('RIVAL BANKED ' + d.banked);
        Popups.add('RIVAL BANKED ' + d.banked, PAL.crimson, { size: 0.34, vy: 1.0, centre: true, life: 1.3 });
      });

      s.on('matchEnd', function (d) { UI.showEnd(d); });
      s.on('rematchPending', function () { UI.toast('RIVAL WANTS A REMATCH'); });
      s.on('opponentLeft', function () { Game.ghost.active = false; UI.toast('RIVAL LEFT THE MATCH'); });
    },

    syncClock: function () {
      var self = this;
      for (var i = 0; i < 5; i++) {
        (function (n) {
          setTimeout(function () {
            if (self.socket && self.socket.connected) self.socket.emit('tsync', Date.now());
          }, n * 140);
        })(i);
      }
    },

    emit: function (ev, data) {
      if (Game.solo || !this.socket || !this.socket.connected) return;
      this.socket.emit(ev, data);
    },

    /** 20 Hz stream: tile x/y, pending score, running flag. */
    startStream: function () {
      var self = this;
      clearInterval(this.timer);
      if (Game.solo) return;
      this.timer = setInterval(function () { self.sendState(); }, 1000 / CFG.NET_HZ);
    },

    stopStream: function () { clearInterval(this.timer); this.timer = 0; },

    sendState: function (force) {
      if (Game.solo || !this.socket || !this.socket.connected) return;
      if (!force && Game.phase !== 'playing' && Game.phase !== 'countdown') return;
      this.socket.emit('state', {
        x: Math.round(Player.fx * 100) / 100,
        y: Math.round(Player.frow * 100) / 100,
        p: Player.pending,
        a: Player.isRunning() ? 1 : 0
      });
    }
  };

      var rival = (d.opponent && d.opponent.name) || Game.rivalName || 'RIVAL';
      var myBanked = d.you.banked;
      var theirBanked = d.opponent ? d.opponent.banked : 0;

      var banner = $('result-banner');
      banner.className = 'result-banner ' + d.result;
      banner.textContent = d.result === 'win' ? 'VICTORY' : d.result === 'loss' ? 'DEFEAT' : 'DRAW';

      $('result-reason').textContent = {
        'both-resolved': myBanked === 0 && theirBanked === 0 ? 'NOBODY BANKED A THING' : 'BOTH RUNS RESOLVED',
        'opponent-left': 'RIVAL LEFT THE MATCH'
      }[d.reason] || '';

      $('end-you-name').textContent = d.you.name || 'YOU';
      $('end-you-score').textContent = myBanked;
      $('end-opp-name').textContent = rival;
      $('end-opp-score').textContent = theirBanked;

      var yf = $('end-you-fate');
      yf.textContent = d.you.status === 'banked' ? 'BANKED' : 'WIPED OUT';
      yf.className = 'h2h-fate ' + (d.you.status === 'banked' ? 'banked' : 'dead');
      var of = $('end-opp-fate');
      if (d.opponent) {
        of.textContent = d.opponent.status === 'banked' ? 'BANKED' : 'WIPED OUT';
        of.className = 'h2h-fate ' + (d.opponent.status === 'banked' ? 'banked' : 'dead');
      } else { of.textContent = 'LEFT'; of.className = 'h2h-fate dead'; }

      $('end-tiles').textContent = Player.maxRow;

      // Fold this match into the lifetime record exactly once.
      if (!Game.recorded) {
        Game.recorded = true;
        H2H.record(rival, d.result, myBanked, theirBanked);
      }
      $('end-best').textContent = H2H.allTimeBest();
      this.renderH2H('end', rival, d.you.name);

      if (d.result === 'win') SFX.cash(); else SFX.blip(220, 0.25, 'sawtooth', 0.05, 120);

      $('btn-again').removeAttribute('disabled');
      $('rematch-note').classList.add('hidden');
      this.hideHud();
      this.show('screen-end');
    },

    showEndSolo: function () {
      Game.phase = 'ended';
      var banked = Player.banked;
      var banner = $('result-banner');
      banner.className = 'result-banner ' + (banked > 0 ? 'win' : 'loss');
      banner.textContent = banked > 0 ? 'CASHED OUT' : 'WIPED OUT';
      $('result-reason').textContent = 'PRACTICE RUN';

      $('end-you-name').textContent = H2H.read('cr_name', '') || 'YOU';
      $('end-you-score').textContent = banked;
      $('end-opp-name').textContent = 'ALL-TIME';
      $('end-opp-score').textContent = H2H.allTimeBest();

      var yf = $('end-you-fate');
      yf.textContent = Player.status === 'banked' ? 'BANKED' : 'WIPED OUT';
      yf.className = 'h2h-fate ' + (Player.status === 'banked' ? 'banked' : 'dead');
      $('end-opp-fate').textContent = 'BEST';
      $('end-opp-fate').className = 'h2h-fate banked';

      $('end-tiles').textContent = Player.maxRow;
      $('end-best').textContent = H2H.allTimeBest();
      this.renderH2H('end', H2H.lastRival());

      this.hideHud();
      this.show('screen-end');
    }
  };

  /* ====================================================================== */
  /* Boot                                                                    */
  /* ====================================================================== */

  // A menu-time idle world so the canvas is never an empty void behind the UI.
  World.reset('MENU-' + ((Math.random() * 1e9) | 0));
  Player.reset();
  Game.cam.row = 3;
  Game.t = 0;

  UI.init();
  Net.connect();
  requestAnimationFrame(loop);

  // Menu ambience: let the world clock run so traffic drifts behind the logo.
  setInterval(function () {
    if (Game.phase === 'menu') {
      Game.t += 0.05;
      Game.cam.row += 0.004;
      World.ensure(Math.ceil(Game.cam.row) + 20);
    }
  }, 50);

  window.addEventListener('pagehide', function () { Net.stopStream(); });
  document.addEventListener('visibilitychange', function () {
    if (!document.hidden) { Game.lastFrame = 0; resize(); }
  });

  // Debug handle â€” lets you poke at the sim from the console without rebuilding.
  window.__CR = {
    CFG: CFG, Game: Game, Player: Player, World: World, Net: Net, UI: UI,
    View: View, H2H: H2H, SFX: SFX, Popups: Popups,
    move: tryMove, bank: bankScore, kill: onDeath,
    itemX: itemX, trainState: trainState, logUnder: logUnder, platformUnder: platformUnder,
    step: function (dt) { update(dt || 0.016); render(); }
  };

/**
 * Generates the home-screen icons (PNG) with zero dependencies â€” Node's zlib
 * is all a PNG needs. Run once: `node tools/make-icons.js`.
 *
 * The icon is the cash-out egg on the score-plate teal, so the app tile on an
 * iPhone matches the two things the player looks at most.
 */
'use strict';

const fs = require('fs');
const path = require('path');
const zlib = require('zlib');

const CRC_TABLE = (() => {
  const t = new Uint32Array(256);
  for (let n = 0; n < 256; n++) {
    let c = n;
    for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;
    t[n] = c >>> 0;
  }
  return t;
})();

function crc32(buf) {
  let c = 0xffffffff;
  for (let i = 0; i < buf.length; i++) c = CRC_TABLE[(c ^ buf[i]) & 0xff] ^ (c >>> 8);
  return (c ^ 0xffffffff) >>> 0;
}

function chunk(type, data) {
  const len = Buffer.alloc(4); len.writeUInt32BE(data.length);
  const td = Buffer.concat([Buffer.from(type, 'ascii'), data]);
  const crc = Buffer.alloc(4); crc.writeUInt32BE(crc32(td));
  return Buffer.concat([len, td, crc]);
}

function encodePNG(w, h, rgba) {
  const raw = Buffer.alloc((w * 4 + 1) * h);
  for (let y = 0; y < h; y++) {
    raw[y * (w * 4 + 1)] = 0;                       // filter: none
    rgba.copy(raw, y * (w * 4 + 1) + 1, y * w * 4, (y + 1) * w * 4);
  }
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(w, 0); ihdr.writeUInt32BE(h, 4);
  ihdr[8] = 8; ihdr[9] = 6; ihdr[10] = 0; ihdr[11] = 0; ihdr[12] = 0;
  return Buffer.concat([
    Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]),
    chunk('IHDR', ihdr),
    chunk('IDAT', zlib.deflateSync(raw, { level: 9 })),
    chunk('IEND', Buffer.alloc(0))
  ]);
}

function mix(a, b, t) { return a + (b - a) * t; }

/** Signed distance-ish helpers so edges are anti-aliased, not jagged. */
function coverage(d, soft) { return Math.max(0, Math.min(1, 0.5 - d / soft)); }

function renderIcon(size) {
  const px = Buffer.alloc(size * size * 4);
  const cx = size / 2, cy = size / 2;
  const R = size * 0.5;
  const soft = Math.max(1, size / 180);

  // Egg geometry: an ellipse whose top is narrower than its bottom.
  const eggRx = size * 0.20, eggRy = size * 0.27, eggCy = cy + size * 0.02;

  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      const i = (y * size + x) * 4;
      const dx = x + 0.5 - cx, dy = y + 0.5 - cy;

      // Plate background with a subtle top-to-bottom shade
      let r = 47, g = 85, b = 102;
      const shadeT = y / size;
      r = mix(r + 10, r - 8, shadeT); g = mix(g + 10, g - 8, shadeT); b = mix(b + 10, b - 8, shadeT);

      // Cream ring inset from the edge (iOS masks the corners itself)
      const dist = Math.sqrt(dx * dx + dy * dy);
      const ringOuter = R * 0.98, ringInner = R * 0.88;
      const ringCov = coverage(Math.abs(dist - (ringOuter + ringInner) / 2) - (ringOuter - ringInner) / 2, soft);
      r = mix(r, 243, ringCov); g = mix(g, 234, ringCov); b = mix(b, 216, ringCov);

      // Egg
      const ey = (y + 0.5 - eggCy) / eggRy;
      const taper = 1 - 0.18 * Math.max(0, -ey);            // narrower up top
      const ex = (x + 0.5 - cx) / (eggRx * taper);
      const eggD = (Math.sqrt(ex * ex + ey * ey) - 1) * eggRy;
      const eggCov = coverage(eggD, soft * 1.2);
      if (eggCov > 0) {
        // Warm white with a highlight up-left and a shadow down-right
        const hx = (x + 0.5 - (cx - eggRx * 0.35)) / eggRx;
        const hy = (y + 0.5 - (eggCy - eggRy * 0.40)) / eggRy;
        const hl = Math.max(0, 1 - Math.sqrt(hx * hx + hy * hy) * 0.9);
        const sh = Math.max(0, (ex * 0.6 + ey * 0.8));
        let er = mix(244, 255, hl), eg = mix(237, 255, hl), eb = mix(224, 255, hl);
        er = mix(er, 205, sh * 0.45); eg = mix(eg, 190, sh * 0.45); eb = mix(eb, 165, sh * 0.45);
        // Ink outline around the egg
        const outline = coverage(Math.abs(eggD) - soft * 2.2, soft);
        er = mix(er, 26, outline * 0.8); eg = mix(eg, 35, outline * 0.8); eb = mix(eb, 44, outline * 0.8);
        r = mix(r, er, eggCov); g = mix(g, eg, eggCov); b = mix(b, eb, eggCov);
      }

      px[i] = Math.round(r); px[i + 1] = Math.round(g); px[i + 2] = Math.round(b); px[i + 3] = 255;
    }
  }
  return encodePNG(size, size, px);
}

/**
 * Chicken Run â€” server
 * Express static host + Socket.io room / match state manager for private 1v1
 * head-to-head matches joined by room code.
 *
 * Design notes:
 *  - The world is generated CLIENT-side from a shared seed the server hands out,
 *    so both players hop through an identical map without streaming any terrain.
 *  - Obstacle positions are pure functions of match time, so a clock handshake
 *    ("tsync") is all that's needed to keep both clients' traffic in lockstep.
 *  - Players never physically interact. There is no cross-player collision:
 *    each runs their own copy of the same layout, and the rival is drawn purely
 *    as a translucent ghost.
 *  - Scoring is pending-vs-banked. A run resolves exactly once, either by
 *    CASHOUT (pending is banked) or DEATH (banked is zero). The server is the
 *    authority on that resolution and on the match result.
 *  - No stakes, entry fees, or wagering of any kind â€” this is a private
 *    for-bragging-rights scoreboard between two people.
 */
'use strict';

const path = require('path');
const http = require('http');
const express = require('express');
const { Server } = require('socket.io');

const PORT = process.env.PORT || 3000;

const app = express();
const server = http.createServer(app);
const io = new Server(server, {
  cors: { origin: '*' },
  pingInterval: 10000,
  pingTimeout: 20000,
  maxHttpBufferSize: 1e5
});

// maxAge 0 + etag => cheap 304s on revalidation, and no stale client after a
// deploy (the asset filenames are unversioned, so caching them would sting).
app.use(express.static(path.join(__dirname, 'public'), {
  maxAge: 0,
  etag: true,
  lastModified: true
}));

// The Netlify-hosted client pings this cross-origin to wake a sleeping Render
// dyno before the player taps CREATE, so allow it explicitly.
app.get('/health', (req, res) => {
  res.set('Access-Control-Allow-Origin', '*');
  res.set('Cache-Control', 'no-store');
  res.json({ ok: true, rooms: rooms.size, sockets: io.engine.clientsCount });
});

/* ------------------------------------------------------------------ */
/* Room registry                                                       */
/* ------------------------------------------------------------------ */

const CODE_ALPHABET = 'ABCDEFGHJKLMNPQRSTUVWXYZ'; // no I/O â€” unambiguous on small screens
const COUNTDOWN_MS = 3500;
const ROOM_TTL_MS = 30 * 60 * 1000;

/** @type {Map<string, object>} */
const rooms = new Map();
/** @type {Map<string, string>} socket.id -> room code */
const socketRoom = new Map();

function makeCode() {
  let code;
  do {
    code = '';
    for (let i = 0; i < 4; i++) code += CODE_ALPHABET[(Math.random() * CODE_ALPHABET.length) | 0];
  } while (rooms.has(code));
  return code;
}

function makeSeed() {
  return (
    Math.random().toString(36).slice(2, 10) + Math.random().toString(36).slice(2, 6)
  ).toUpperCase();
}

function sanitizeName(n) {
  if (typeof n !== 'string') return '';
  return n.replace(/[^\w \-]/g, '').trim().slice(0, 12).toUpperCase();
}

function newPlayer(socket, slot, name) {
  return {
    id: socket.id,
    slot,                       // 0 = host, 1 = challenger (drives avatar colour)
    name: sanitizeName(name) || (slot === 0 ? 'PLAYER 1' : 'PLAYER 2'),
    x: 0,
    y: 0,
    pending: 0,                 // live, at-risk score
    banked: 0,                  // locked-in score; stays 0 if they die
    status: 'running',          // running | banked | dead
    connected: true,
    wantsRematch: false
  };
}

function createRoom(socket, name) {
  const code = makeCode();
  const room = {
    code,
    seed: makeSeed(),
    state: 'lobby',            // lobby | countdown | playing | ended
    startAt: 0,
    createdAt: Date.now(),
    touchedAt: Date.now(),
    players: [newPlayer(socket, 0, name)]
  };
  rooms.set(code, room);
  socketRoom.set(socket.id, code);
  socket.join(code);
  return room;
}

function roomOf(socket) {
  const code = socketRoom.get(socket.id);
  return code ? rooms.get(code) : null;
}

function playerIn(room, id) {
  return room ? room.players.find((p) => p.id === id) : null;
}

function opponentIn(room, id) {
  return room ? room.players.find((p) => p.id !== id) : null;
}

function lobbyView(room) {
  return {
    code: room.code,
    state: room.state,
    players: room.players.map((p) => ({
      id: p.id, slot: p.slot, name: p.name, connected: p.connected
    }))
  };
}

function pushLobby(room) {
  room.touchedAt = Date.now();
  io.to(room.code).emit('room', lobbyView(room));
}

function clampNum(v, lo, hi) {
  const n = Number(v);
  if (!Number.isFinite(n)) return 0;
  return Math.min(hi, Math.max(lo, n));
}

/* ------------------------------------------------------------------ */
/* Match lifecycle                                                     */
/* ------------------------------------------------------------------ */

function startMatch(room) {
  if (room.players.length < 2) return;
  room.seed = makeSeed();
  room.state = 'countdown';
  room.startAt = Date.now() + COUNTDOWN_MS;
  room.touchedAt = Date.now();

  for (const p of room.players) {
    p.x = 0; p.y = 0;
    p.pending = 0; p.banked = 0;
    p.status = 'running';
    p.wantsRematch = false;
  }

  for (const p of room.players) {
    const opp = opponentIn(room, p.id);
    io.to(p.id).emit('matchStart', {
      seed: room.seed,
      startAt: room.startAt,
      you: { slot: p.slot, name: p.name },
      opponent: opp ? { slot: opp.slot, name: opp.name } : null
    });
  }

  // Flip to "playing" the moment the gate opens so late joiners can't slip in.
  setTimeout(() => {
    if (rooms.get(room.code) === room && room.state === 'countdown') room.state = 'playing';
  }, COUNTDOWN_MS);
}

/** A match is over once both runs have resolved (banked or dead). */
function matchIsOver(room) {
  return room.players.every((p) => p.status !== 'running');
}

function endMatch(room, reason) {
  if (room.state === 'ended' || room.state === 'lobby') return;
  room.state = 'ended';
  room.touchedAt = Date.now();

  for (const p of room.players) {
    const opp = opponentIn(room, p.id);
    let result = 'draw';
    if (!opp || !opp.connected) {
      result = 'win';
    } else if (p.banked > opp.banked) {
      result = 'win';
    } else if (p.banked < opp.banked) {
      result = 'loss';
    }
    io.to(p.id).emit('matchEnd', {
      reason,
      result,
      you: { name: p.name, slot: p.slot, banked: p.banked, status: p.status },
      opponent: opp
        ? {
            name: opp.name, slot: opp.slot, banked: opp.banked,
            status: opp.status, connected: opp.connected
          }
        : null
    });
  }
}

/* ------------------------------------------------------------------ */
/* Socket wiring                                                       */
/* ------------------------------------------------------------------ */

io.on('connection', (socket) => {
  // Clock handshake â€” clients average a few of these to align match time.
  socket.on('tsync', (t0) => {
    socket.emit('tsync', { t0: Number(t0) || 0, server: Date.now() });
  });

  socket.on('createMatch', (payload, ack) => {
    if (roomOf(socket)) leaveRoom(socket);
    const room = createRoom(socket, payload && payload.name);
    if (typeof ack === 'function') ack({ ok: true, code: room.code, slot: 0 });
    pushLobby(room);
  });

  socket.on('joinMatch', (payload, ack) => {
    const code = String((payload && payload.code) || '')
      .toUpperCase().replace(/[^A-Z]/g, '').slice(0, 4);
    const room = rooms.get(code);
    const reply = (o) => { if (typeof ack === 'function') ack(o); };

    if (!room) return reply({ ok: false, error: 'NO MATCH WITH THAT CODE' });
    if (room.players.length >= 2) return reply({ ok: false, error: 'THAT MATCH IS FULL' });
    if (room.state !== 'lobby' && room.state !== 'ended') {
      return reply({ ok: false, error: 'MATCH ALREADY IN PROGRESS' });
    }

    if (roomOf(socket)) leaveRoom(socket);
    const player = newPlayer(socket, 1, payload && payload.name);
    room.players.push(player);
    socketRoom.set(socket.id, code);
    socket.join(code);

    reply({ ok: true, code: room.code, slot: 1 });
    pushLobby(room);
    startMatch(room);
  });

  // 20 Hz stream: tile x/y, live pending score, still-running flag.
  socket.on('state', (d) => {
    const room = roomOf(socket);
    if (!room || (room.state !== 'playing' && room.state !== 'countdown')) return;
    const me = playerIn(room, socket.id);
    if (!me || !d) return;

    me.x = clampNum(d.x, -32, 32);
    me.y = clampNum(d.y, -32, 100000);
    me.pending = clampNum(d.p, 0, 10000000) | 0;
    room.touchedAt = Date.now();

    // NOTE: run status is deliberately NOT taken from this stream. The client
    // force-sends one last packet (a:0) immediately before it emits `died` or
    // `cashout`, so trusting it here would flip the status first and make those
    // handlers' duplicate guards swallow the real resolution â€” ending the match
    // never. Lifecycle comes from `died`/`cashout` only; `a` is relay-only.
    const opp = opponentIn(room, socket.id);
    if (opp) {
      io.to(opp.id).emit('oppState', { x: me.x, y: me.y, p: me.pending, a: d.a ? 1 : 0 });
    }
  });

  // CASHOUT â€” the whole game. Locks pending into banked and ends that run.
  socket.on('cashout', (d) => {
    const room = roomOf(socket);
    if (!room || room.state !== 'playing') return;
    const me = playerIn(room, socket.id);
    if (!me || me.status !== 'running') return;

    if (d && typeof d.p === 'number') me.pending = clampNum(d.p, 0, 10000000) | 0;
    me.banked = me.pending;
    me.status = 'banked';

    const opp = opponentIn(room, socket.id);
    if (opp) io.to(opp.id).emit('oppCashout', { banked: me.banked });
    if (matchIsOver(room)) endMatch(room, 'both-resolved');
  });

  // DEATH â€” pending is forfeit. Nothing is ever auto-saved on death.
  socket.on('died', () => {
    const room = roomOf(socket);
    if (!room || room.state !== 'playing') return;
    const me = playerIn(room, socket.id);
    if (!me || me.status !== 'running') return;

    me.pending = 0;
    me.banked = 0;
    me.status = 'dead';

    const opp = opponentIn(room, socket.id);
    if (opp) io.to(opp.id).emit('oppDied', {});
    if (matchIsOver(room)) endMatch(room, 'both-resolved');
  });

  socket.on('rematch', () => {
    const room = roomOf(socket);
    if (!room || room.state !== 'ended') return;
    const me = playerIn(room, socket.id);
    if (!me) return;
    me.wantsRematch = true;

    const opp = opponentIn(room, socket.id);
    if (opp && opp.connected && opp.wantsRematch) {
      startMatch(room);
    } else if (opp && opp.connected) {
      io.to(opp.id).emit('rematchPending', { name: me.name });
    }
  });

  socket.on('leaveMatch', () => leaveRoom(socket));
  socket.on('disconnect', () => leaveRoom(socket, true));

  function leaveRoom(sock, disconnected) {
    const code = socketRoom.get(sock.id);
    if (!code) return;
    socketRoom.delete(sock.id);
    const room = rooms.get(code);
    sock.leave(code);
    if (!room) return;

    const idx = room.players.findIndex((p) => p.id === sock.id);
    if (idx === -1) return;
    room.players.splice(idx, 1);
    room.touchedAt = Date.now();

    if (room.players.length === 0) {
      rooms.delete(code);
      return;
    }

    io.to(code).emit('opponentLeft', { disconnected: !!disconnected });
    if (room.state === 'countdown' || room.state === 'playing') {
      endMatch(room, 'opponent-left');
    } else {
      room.state = 'lobby';
      // Promote the remaining player to host so the room code stays live.
      room.players[0].slot = 0;
      room.players[0].wantsRematch = false;
      pushLobby(room);
    }
  }
});

// Sweep abandoned rooms.
setInterval(() => {
  const now = Date.now();
  for (const [code, room] of rooms) {
    if (room.players.length === 0 || now - room.touchedAt > ROOM_TTL_MS) rooms.delete(code);
  }
}, 60000).unref();

server.listen(PORT, () => {
  console.log('Chicken Run running at http://localhost:' + PORT);
});
